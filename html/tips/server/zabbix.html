sudo nano /etc/zabbix/zabbix_agentd.conf.d/device_performance.conf

sudo systemctl restart zabbix-agent

# =============================================================================
# .env  -  Zabbix Docker 環境設定ファイル
#
# バージョンアップ時は ZABBIX_VERSION のみ変更すればOK
# 起動コマンド: docker compose up --build -d

# ---------- コンテナ名 ----------
ZABBIX_CONTAINER_NAME=Zabbix_Server
DB_CONTAINER_NAME=Zabbix_DB
WEB_CONTAINER_NAME=Zabbix_Web

# docker-compose.yml からの相対パス、または絶対パス

# ---------- Zabbix イメージバージョン ----------

# ---------- MySQL イメージバージョン ----------
MYSQL_VERSION=8.0

# ---------- DB 接続情報 ----------

MYSQL_PASSWORD=zabbixpass

# ---------- Zabbix サーバ ----------
ZBX_SERVER_HOST=zabbix-server

# ---------- Web UI ----------
PHP_TZ=Asia/Tokyo

# ---------- Zabbix トラッパーポート ----------

services:

  zabbix-db:
    image: mysql:${MYSQL_VERSION}
    container_name: ${DB_CONTAINER_NAME}
    env_file: .env

    environment:
      MYSQL_DATABASE:       ${MYSQL_DATABASE}
      MYSQL_USER:           ${MYSQL_USER}
      MYSQL_PASSWORD:       ${MYSQL_PASSWORD}
      MYSQL_ROOT_PASSWORD:  ${MYSQL_ROOT_PASSWORD}

    command:
      - --max_allowed_packet=64M
      - --explicit_defaults_for_timestamp=1
      - --log-bin-trust-function-creators=1

    volumes:
      - ./zabbix-db:/var/lib/mysql

    restart: unless-stopped

    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost", "-u$$MYSQL_USER", "-p$$MYSQL_PASSWORD"]
      interval: 5s
      timeout: 5s
      retries: 10


  zabbix-server:
    build:
      context: ${DOCKERFILE_DIR}            # .env で指定するビルドコンテキスト
      dockerfile: ${DOCKERFILE}             # .env で指定する Dockerfile パス
      args:
        ZABBIX_VERSION: ${ZABBIX_VERSION}   # FROM の ARG に渡してバージョンを動的に変える
        ENTRY_POINT:    ${ENTRY_POINT}
    image: zabbix-server-custom:${ZABBIX_VERSION}   # ビルド後イメージ名 (キャッシュ識別用)
    container_name: ${ZABBIX_CONTAINER_NAME}
    env_file: .env

    environment:
      DB_SERVER_HOST:   ${DB_SERVER_HOST}
      DB_SERVER_PORT:   ${DB_SERVER_PORT}
      MYSQL_DATABASE:   ${MYSQL_DATABASE}
      MYSQL_USER:       ${MYSQL_USER}
      MYSQL_PASSWORD:   ${MYSQL_PASSWORD}
      ZBX_CREATE_SQL:   ${ZBX_CREATE_SQL}

    ports:
      - "${ZBX_TRAPPER_PORT}:10051"

    depends_on:
      zabbix-db:
        condition: service_healthy          # DB healthcheck パス後に起動

    volumes:
      - ./zabbix_server.conf:/etc/zabbix/zabbix_server.conf

    restart: unless-stopped

    command:
      - /usr/local/bin/${ENTRY_POINT}


  zabbix-web:
    image: zabbix/zabbix-web-nginx-mysql:${ZABBIX_VERSION}
    container_name: ${WEB_CONTAINER_NAME}
    env_file: .env

    environment:
      DB_SERVER_HOST:   ${DB_SERVER_HOST}
      MYSQL_USER:       ${MYSQL_USER}
      MYSQL_PASSWORD:   ${MYSQL_PASSWORD}
      MYSQL_DATABASE:   ${MYSQL_DATABASE}
      ZBX_SERVER_HOST:  ${ZBX_SERVER_HOST}
      PHP_TZ:           ${PHP_TZ}

    ports:
      - "${WEB_PORT}:8080"

    depends_on:
      zabbix-server:
        condition: service_started
      zabbix-db:
        condition: service_healthy

    restart: unless-stopped

# =============================================================================
# zabbix-init/Dockerfile
#
# zabbix-server-mysql 公式イメージをベースに、
# DB 自動初期化エントリーポイントを追加する薄いラッパー。
#
# イメージ本体は変更せず entrypoint だけ差し替えるため、
# Zabbix バージョンアップ時は .env の ZABBIX_VERSION を変えるだけでよい。
# =============================================================================

ARG ZABBIX_VERSION
FROM zabbix/zabbix-server-mysql:${ZABBIX_VERSION}

USER root

ARG ENTRY_POINT

COPY ${ENTRY_POINT} /usr/local/bin/${ENTRY_POINT}
RUN chmod +x /usr/local/bin/${ENTRY_POINT}

# 公式イメージのデフォルトユーザに戻す
USER 1997

start.sh 
#!/bin/bash
# =============================================================================
# entrypoint.sh  -  Zabbix Server カスタムエントリーポイント
#
# 役割:
#   1. MySQL が応答するまで待機
#   2. Zabbix テーブルが存在しない(初回)場合のみ schema / data を投入
#   3. 本来の zabbix_server エントリーポイントへ exec で引き渡す
#
# ※ このスクリプトは zabbix-server コンテナ内で動作する
#    DB 接続情報は docker-compose.yml の environment / .env から渡される
# =============================================================================

set -euo pipefail

log() { echo "[entrypoint $(date '+%H:%M:%S')] $*"; }

# ---------- 環境変数 (docker-compose.yml の environment で注入される) ----------
: "${DB_SERVER_HOST:?DB_SERVER_HOST が未設定です}"
: "${DB_SERVER_PORT:=3306}"
: "${MYSQL_DATABASE:?MYSQL_DATABASE が未設定です}"
: "${MYSQL_USER:?MYSQL_USER が未設定です}"
: "${MYSQL_PASSWORD:?MYSQL_PASSWORD が未設定です}"

# SQLファイルは .env の ZBX_CREATE_SQL で変更可能
# コンマ区切りで複数ファイルを指定できる
# 例(単一): ZBX_CREATE_SQL=/usr/share/doc/zabbix-server-mysql/create.sql.gz
# 例(複数): ZBX_CREATE_SQL=schema.sql.gz,data.sql.gz,custom.sql.gz
: "${ZBX_CREATE_SQL:=/usr/share/doc/zabbix-server-mysql/create.sql.gz}"
MYSQL_OPTS="--ssl=false"
MAX_WAIT=30

# ---------- MySQL 起動待ち ----------
log "MySQL (${DB_SERVER_HOST}:${DB_SERVER_PORT}) の起動を待ちます..."
count=0
until mysqladmin ping ${MYSQL_OPTS} \
        -h "${DB_SERVER_HOST}" \
        -P "${DB_SERVER_PORT}" \
        -u "${MYSQL_USER}" \
        -p"${MYSQL_PASSWORD}" \
        --silent 2>/dev/null; do
    count=$((count + 1))
    if [ "${count}" -ge "${MAX_WAIT}" ]; then
        log "ERROR: MySQL が応答しません (${MAX_WAIT}回試行)。終了します。"
        exit 1
    fi
    log "待機中... (${count}/${MAX_WAIT})"
    sleep 2
done
log "MySQL 接続OK。"

# ---------- 初回判定 ----------
TABLE_EXISTS=$(mysql ${MYSQL_OPTS} \
    -h "${DB_SERVER_HOST}" \
    -P "${DB_SERVER_PORT}" \
    -u "${MYSQL_USER}" \
    -p"${MYSQL_PASSWORD}" \
    "${MYSQL_DATABASE}" \
    --skip-column-names -e \
    "SELECT COUNT(*) FROM information_schema.tables
     WHERE table_schema='${MYSQL_DATABASE}' AND table_name='users';" \
    2>/dev/null || echo "0")

if [ "${TABLE_EXISTS}" = "0" ] || [ -z "${TABLE_EXISTS}" ]; then
    log "初回起動 — DB を初期化します..."

    # コンマ区切りで複数の SQL ファイルを順番に適用する
    OLD_IFS="$IFS"
    IFS=','
    sql_index=0
    for sql_file in ${ZBX_CREATE_SQL}; do
        # 前後の空白・タブを除去
        sql_file="$(echo "${sql_file}" | tr -d ' \t')"
        [ -z "${sql_file}" ] && continue
        sql_index=$((sql_index + 1))

        if [ ! -f "${sql_file}" ]; then
            log "ERROR: SQL ファイルが見つかりません: ${sql_file}"
            IFS="$OLD_IFS"
            exit 1
        fi

        log "SQL を適用中... [${sql_index}] ${sql_file}"
        zcat "${sql_file}" | mysql ${MYSQL_OPTS} \
            -h "${DB_SERVER_HOST}" \
            -P "${DB_SERVER_PORT}" \
            -u "${MYSQL_USER}" \
            -p"${MYSQL_PASSWORD}" \
            "${MYSQL_DATABASE}"
    done
    IFS="$OLD_IFS"

    if [ "${sql_index}" -eq 0 ]; then
        log "ERROR: ZBX_CREATE_SQL に有効なファイルが1件もありません。"
        exit 1
    fi

    log "DB 初期化完了 (${sql_index} ファイル適用)。"
else
    log "既存 DB を検出 — 初期化をスキップします。"
fi

# ---------- 本来の Zabbix Server エントリーポイントへ ----------
log "Zabbix Server を起動します..."
exec /usr/sbin/zabbix_server --foreground "$@"

Zabbixプロキシの構築 (Docker推奨)

.env 
# =============================================================================
# # .env - Zabbix Proxy (Docker) 環境設定ファイル
#
# 起動コマンド: `docker compose up -d`
#
# ## 最重要ルール
#
# `ZABBIX_VERSION` は **Zabbix_Server/.env と必ず同じ値** にすること。
# プロキシはサーバと「メジャー.マイナー」(例: 7.4) を揃える必要がある。
# バージョンアップ時はサーバ → プロキシの順で、両方の `ZABBIX_VERSION` を更新する。
#
# ## 起動後にやること(初回のみ)
#
# 1. サーバの Web UI で `管理 > プロキシ` を開き、プロキシを登録する
# 2. 登録名を `ZBX_HOSTNAME` と **完全に一致** させる(大文字小文字も区別される)
# 3. `ZBX_PROXYMODE` に合わせて動作モードを選ぶ
#
# 注意: インラインコメント(値の後ろの `#`)は意図的に使っていない。
#       値の末尾に空白やコメントが混入する事故を避けるため。
# =============================================================================


# -----------------------------------------------------------------------------
# ## コンテナ名
#
# | 変数 | 役割 |
# |------|------|
# | `PROXY_CONTAINER_NAME` | Zabbix Proxy コンテナ |
# | `DB_CONTAINER_NAME`    | プロキシ用 MySQL コンテナ |
# -----------------------------------------------------------------------------
PROXY_CONTAINER_NAME=Zabbix_Proxy
DB_CONTAINER_NAME=Zabbix_Proxy_DB


# -----------------------------------------------------------------------------
# ## イメージバージョン
#
# | 変数 | 説明 | 備考 |
# |------|------|------|
# | `ZABBIX_VERSION` | `zabbix/zabbix-proxy-mysql` のタグ | **サーバと同じ値にする** |
# | `MYSQL_VERSION`  | `mysql` のタグ | プロキシ専用 DB のため、サーバ側と異なっていてもよい |
#
# タグ一覧: https://hub.docker.com/r/zabbix/zabbix-proxy-mysql/tags
# -----------------------------------------------------------------------------
ZABBIX_VERSION=alpine-7.4-latest
MYSQL_VERSION=8.4


# -----------------------------------------------------------------------------
# ## プロキシ動作設定
#
# | 変数 | 既定値 | 説明 |
# |------|--------|------|
# | `ZBX_PROXYMODE` | `0` | `0` = アクティブ(プロキシ → サーバへ接続) / `1` = パッシブ(サーバ → プロキシへ接続) |
# | `ZBX_HOSTNAME`  | `Zabbix_Proxy` | プロキシ名。Web UI の登録名と完全一致させる(使用可: 英数字 `.` `_` `-`) |
# | `ZBX_SERVER_HOST` | (**必須**) | サーバのアドレス。`host:port` 形式。ポート省略時は 10051 |
# | `ZBX_PROXY_PORT` | `10051` | ホスト側で公開するポート |
# | `ZBX_PROXYOFFLINEBUFFER` | `24` | サーバ切断時にデータを保持する時間 (時間単位) |
# | `ZBX_PROXYLOCALBUFFER`   | `0`  | サーバへ送信済みデータをローカルに保持する時間 (時間単位) |
# | `ZBX_DEBUGLEVEL` | `3` | ログレベル (0=なし 〜 5=デバッグ) |
#
# ### モードごとの `ZBX_SERVER_HOST` の意味
#
# | モード | `ZBX_SERVER_HOST` の意味 | 必要な通信 |
# |--------|--------------------------|-----------|
# | アクティブ (`0`) | 接続先サーバのアドレス | プロキシ → サーバ `10051/TCP` |
# | パッシブ (`1`)   | 接続を許可するサーバの IP | サーバ → プロキシ `ZBX_PROXY_PORT/TCP` |
#
# ### 同一ホストでサーバと同居させる場合
#
# サーバが `10051` を公開済みのため、`ZBX_PROXY_PORT` を `10061` などに変更すること。
# `ZBX_SERVER_HOST` には、コンテナ名ではなく **ホストの IP** を指定する
# (サーバ側 compose とはネットワークが別のため)。
#
# ### 注意
#
# `ZBX_SERVER_PORT` は Zabbix 6.0 以降で廃止されたため使用しない。
# ポートは `ZBX_SERVER_HOST` に `:ポート番号` を付けて指定する。
# -----------------------------------------------------------------------------
ZBX_PROXYMODE=0
ZBX_HOSTNAME=Zabbix_Proxy
ZBX_SERVER_HOST=192.168.10.64:10051
ZBX_PROXY_PORT=10051
ZBX_PROXYOFFLINEBUFFER=24
ZBX_PROXYLOCALBUFFER=0
ZBX_DEBUGLEVEL=3


# -----------------------------------------------------------------------------
# ## DB 接続情報 (プロキシ専用 DB)
#
# サーバ側の DB とは **完全に別物** DB 名・パスワードも独立して設定してよい。
#
# | 変数 | 説明 |
# |------|------|
# | `MYSQL_DATABASE`      | プロキシ用 DB 名 |
# | `MYSQL_USER`          | Zabbix Proxy が接続するユーザ |
# | `MYSQL_PASSWORD`      | 同上のパスワード (**本番では必ず変更**) |
# | `MYSQL_ROOT_PASSWORD` | MySQL root パスワード (**本番では必ず変更**) |
# | `DB_SERVER_HOST`      | DB のホスト名 compose のサービス名 `zabbix-proxy-db` と一致させる |
# | `DB_SERVER_PORT`      | DB のポート |
# -----------------------------------------------------------------------------
MYSQL_DATABASE=zabbix_proxy
MYSQL_USER=zabbix
MYSQL_PASSWORD=proxypass
MYSQL_ROOT_PASSWORD=proxyrootpass
DB_SERVER_HOST=zabbix-proxy-db
DB_SERVER_PORT=3306

compose.yml 
# =============================================================================
# # docker-compose.yml - Zabbix Proxy (MySQL)
#
# ## 構成
#
# | サービス | イメージ | 役割 |
# |----------|----------|------|
# | `zabbix-proxy-db` | `mysql` | プロキシ専用 DB (収集データのバッファ) |
# | `zabbix-proxy`    | `zabbix/zabbix-proxy-mysql` | Zabbix Proxy 本体 |
#
# ## 設計メモ
#
# - 公式イメージの標準エントリーポイントが、初回起動時に DB スキーマを自動作成する。
#   そのためサーバ側のようなカスタム `Dockerfile` / `entry.sh` は不要。
# - バージョンは `.env` の `ZABBIX_VERSION` だけで決まる。
# - `env_file` は使わず、必要な変数のみ `environment` で明示的に渡す。
#   (`.env` 全体をコンテナへ流し込むと、無関係な変数が混入するため)
# =============================================================================

services:

  # ---------------------------------------------------------------------------
  # ## zabbix-proxy-db
  #
  # | `command` オプション | 説明 |
  # |----------------------|------|
  # | `--character-set-server=utf8mb4` | 文字コード |
  # | `--collation-server=utf8mb4_bin` | Zabbix 推奨の照合順序 |
  # | `--max_allowed_packet=64M`       | 大きなパケットを許可 |
  # | `--skip-log-bin`                 | バイナリログを無効化 (ディスク肥大化の防止) |
  #
  # ### healthcheck について
  #
  # `-h 127.0.0.1` は **意図的な指定**。`localhost` だとソケット経由になり、
  # MySQL の初回初期化中に立ち上がる一時サーバでも ping が成功してしまう。
  # TCP 経由にすることで、初期化が完全に終わってから healthy になる。
  # ---------------------------------------------------------------------------
  zabbix-proxy-db:
    image: mysql:${MYSQL_VERSION}
    container_name: ${DB_CONTAINER_NAME}

    environment:
      MYSQL_DATABASE:       ${MYSQL_DATABASE}
      MYSQL_USER:           ${MYSQL_USER}
      MYSQL_PASSWORD:       ${MYSQL_PASSWORD}
      MYSQL_ROOT_PASSWORD:  ${MYSQL_ROOT_PASSWORD}

    command:
      - --character-set-server=utf8mb4
      - --collation-server=utf8mb4_bin
      - --max_allowed_packet=64M
      - --skip-log-bin

    volumes:
      - ./zabbix-proxy-db:/var/lib/mysql

    restart: unless-stopped

    healthcheck:
      test: ["CMD-SHELL", "mysqladmin ping -h 127.0.0.1 -u$$MYSQL_USER -p$$MYSQL_PASSWORD --silent"]
      interval: 5s
      timeout: 5s
      retries: 20
      start_period: 20s


  # ---------------------------------------------------------------------------
  # ## zabbix-proxy
  #
  # | 環境変数 | 対応する zabbix_proxy.conf のパラメータ |
  # |----------|----------------------------------------|
  # | `ZBX_PROXYMODE`          | `ProxyMode` |
  # | `ZBX_HOSTNAME`           | `Hostname` |
  # | `ZBX_SERVER_HOST`        | `Server` |
  # | `ZBX_PROXYOFFLINEBUFFER` | `ProxyOfflineBuffer` |
  # | `ZBX_PROXYLOCALBUFFER`   | `ProxyLocalBuffer` |
  # | `ZBX_DEBUGLEVEL`         | `DebugLevel` |
  # | `DB_SERVER_HOST` / `DB_SERVER_PORT` | `DBHost` / `DBPort` |
  # | `MYSQL_DATABASE` / `MYSQL_USER` / `MYSQL_PASSWORD` | `DBName` / `DBUser` / `DBPassword` |
  #

  #

SNMP

  • SNMP(Simple Network Management Protocol)は、ネットワーク機器を 監視・管理するためのプロトコル
  • ルータ、スイッチ、サーバ、プリンタなどの状態(トラフィック量、CPU使用率、稼働時間など)を取得・設定できる
  • トランスポートは基本的に UDP
  • マネージャ(NMS) が エージェント(監視対象機器) に問い合わせる、またはエージェントがマネージャへ通知する
  • 管理対象の情報は MIB に木構造で定義され、OID で指定する
  • バージョンは v1 / v2c / v3 の3種類
    v1・v2c : 平文
    v3 : 認証・暗号化に対応

ポート

| ポート種別 | ポート番号 | ポート名 | 説明 | |---|---|---|---| | UDP | 161 | snmp | エージェントがマネージャからのリクエスト(Get / GetNext / GetBulk / Set)を受け付ける | | UDP | 162 | snmptrap | マネージャがエージェントからの通知(Trap / Inform)を受け付ける | | TCP / UDP | 10161 | snmpdtls | TLS / DTLS 上の SNMP(リクエスト受付)、RFC 6353 | | TCP / UDP | 10162 | snmpdtls-trap | TLS / DTLS 上の SNMP(通知受付)、RFC 6353 |

通信方法

用語

| 用語 | 説明 | |---|---| | コミュニティ名 |
  • v1 / v2c で使うパスワード代わりの文字列
  • 一致しないとエージェントは応答しない
  • 平文で流れるため、既定値のまま運用するのは危険
コミュニティ名権限説明
public読み取り専用(RO)
  • 読み取り用の既定値として多い
  • 第三者に機器情報を取得される恐れがある
private読み書き(RW)
  • 書き込み用の既定値として多い
  • 設定を書き換えられる恐れがある
| | マネージャ(NMS) |
  • 監視する側
  • エージェントへ問い合わせを行う
  • エージェントからの通知を受信する
| | エージェント |
  • 監視される側
  • 機器上で動作し、情報を提供・通知する
| | MIB |
  • 管理対象オブジェクトの定義集
  • ツリー構造
  • 標準は MIB-II(`1.3.6.1.2.1`)
  • ベンダー独自は enterprises(`1.3.6.1.4.1`)配下
| | OID |
  • MIB のツリー上の位置を `.` 区切りの数字で表したもの
  • 末尾の `.0` はスカラー値を表す
名称OID内容
sysDescr1.3.6.1.2.1.1.1.0機器の説明
sysUpTime1.3.6.1.2.1.1.3.0起動からの経過時間
sysName1.3.6.1.2.1.1.5.0機器名
ifTable1.3.6.1.2.1.2.2インターフェース情報
| | Trap |
  • エージェントから能動的に送る通知
  • 応答がない
  • UDP のため欠落しても検知できない
| | Inform |
  • 応答つきの通知(v2c以降)
  • マネージャが Response を返す
  • 応答がなければ再送される
| | PDU |
  • SNMP のメッセージ単位
PDU方向内容
GetRequestM → A指定OIDの値を取得
GetNextRequestM → A次のOIDの値を取得(walkの基本)
GetBulkRequestM → A複数の値をまとめて取得(v2c以降)
SetRequestM → A値を設定
ResponseA → M各リクエストへの応答
| | バージョン |
バージョン認証暗号化備考
v1コミュニティ名なし古い機器向け
v2cコミュニティ名なしGetBulk / Inform に対応、現在も広く利用
v3ユーザ認証(USM)ありVACM でアクセス範囲も制御、推奨
| | USM / VACM |
  • v3 のセキュリティ機構
  • USM はユーザ認証と暗号化
  • VACM はユーザごとのアクセス制御
|
