DISP_STR(){
    for arg in "$@"; do
        printf "%s" "${arg}"
    done
}

trim() {
    local var="$1"
    var="${var#"${var%%[![:space:]]*}"}"   # 前
    var="${var%"${var##*[![:space:]]}"}"   # 後
    echo "$var"
}

# ===== ログ出力用 =====
LOG(){
    local name=$(trim "${1:-USERAPP}")
    local state=$(trim "${2:-LOG}")
    local action=$(trim "${3:-ACTION}")
    local nowtime="$(date '+%Y/%m/%d %H:%M:%S')"
    local detail=$(trim "${4:-log detail}")
    local header="[${name}][${state}][${action}]"
    local headerSize="${5:-50}"
    
    printf "%-${headerSize}s %s %s\n" "${header}" "${nowtime}" "${detail}"
}

LOG_INFO(){
    local name="${1:-USERAPP}"
    local state="INFO"
    local action="${2:-ACTION}"
    local detail="${3:-detail}"
    local headerSize="${4:-50}"

    DISP_STR "${FG_BRIGHT_CYAN}" "${BG_BLACK}"
    LOG "${name}" "${state}" "${action}" "${detail}" "${headerSize}"
    DISP_STR "${RESET}"
}

# ===== フールプルーフ =====
BLOCKED_COMMANDS=("chmod" "chown" "rm")
BLOCKED_DIRS=("/")
ROOT_CHILD_LIMIT=3

_guard_check() {
    local cmd_name
    cmd_name="$(basename -- "$1")"
    shift

    local blocked=1 c
    for c in "${BLOCKED_COMMANDS[@]}"; do
        [[ "$cmd_name" == "$c" ]] && blocked=0
    done
    (( blocked == 0 )) || return 0

    local arg resolved bd parent
    local end_opts=0 root_children=0

    for arg in "$@"; do
        if [[ "$arg" == "--no-preserve-root" ]]; then
            echo "[Reject] --no-preserve-root は使用禁止 ($cmd_name)" >&2
            return 1
        fi
        if (( end_opts == 0 )); then
            [[ "$arg" == "--" ]] && { end_opts=1; continue; }
            [[ "$arg" == -* ]] && continue
        fi

        # 存在しないパスも正規化できるよう -m を使う (GNU coreutils)
        resolved="$(realpath -m -- "$arg" 2>/dev/null)" || continue

        for bd in "${BLOCKED_DIRS[@]}"; do
            if [[ "$resolved" == "$bd" ]]; then
                echo "[Reject] '$arg' は '$bd' に解決されます (pwd=$PWD, cmd=$cmd_name)" >&2
                return 1
            fi
        done

        # `/*` はシェルが展開してから渡されるため、/ 直下が大量に並ぶ形で検出する
        parent="$(dirname -- "$resolved")"
        if [[ "$parent" == "/" && "$resolved" != "/" && -d "$resolved" ]]; then
            root_children=$((root_children + 1))
        fi
    done

    if (( root_children >= ROOT_CHILD_LIMIT )); then
        echo "[Reject] '/' 直下のディレクトリが ${root_children} 個指定されています (/* 展開の疑い, cmd=$cmd_name)" >&2
        return 1
    fi
    return 0
}

sudo() {
    local args=("$@") i=0
    while (( i < ${#args[@]} )); do
        case "${args[i]}" in
            -u|-g|-C|-D|-h|-p|-R|-r|-T|-t|-U) i=$((i + 2)) ;;
            -*) i=$((i + 1)) ;;
            *) break ;;
        esac
    done
    if (( i < ${#args[@]} )); then
        _guard_check "${args[@]:i}" || return 1
    fi
    command sudo "$@"
}

for _guard_cmd in "${BLOCKED_COMMANDS[@]}"; do
    eval "${_guard_cmd}() { _guard_check ${_guard_cmd} \"\$@\" || return 1; command ${_guard_cmd} \"\$@\"; }"
done
unset _guard_cmd

WHOIS RFC3912

 インターネット上の資源(ドメイン名・IPアドレス・AS番号など)について、誰が登録・管理しているか、いつ登録されたか、どこに連絡できるかといった登録情報を、サーバへ問い合わせて取得するための クエリ/レスポンス型プロトコル
 ドメインの取得可否確認、管理組織・連絡先の確認、ネットワーク障害や不正アクセス時の管理者調査などに利用される

仕組み

  1. クライアントがサーバの TCP 43 番ポートへ接続する
  2. クライアントがクエリ文字列 + CRLF を送信する
  3. サーバが応答(任意のテキスト)を返す
  4. サーバが接続を閉じる(接続終了がレスポンス終端の合図)

インストール方法

ターミナル 
sudo apt install whois

whoisコマンド使い方

ターミナル 
whois [オプション] [ターゲット]

##### オプション一覧 | オプション | 説明 | |------------|------| | `-h [問い合わせ先WHOISサーバ]`, `--host [問い合わせ先WHOISサーバ]` | 問い合わせ先 WHOIS サーバを指定 | | `-p [ポート番号]`, `--port [ポート番号]` | 接続ポートを指定(既定 43) | | `-H` | 法的免責事項(disclaimer)を非表示にする | | `-I` | `whois.iana.org` に問い合わせる | | `-r` | RIPE 系で連絡先情報の表示を抑制 | | `-B` | RIPE 系で IP 範囲をフィルタせず全て表示 | | `-b` | RIPE 系で IP 範囲のみ簡易表示 | | `-i [属性]` | RIPE 系の逆引きルックアップ | | `-T [オブジェクト種別]` | RIPE 系で検索するオブジェクト種別を限定 | | `-t [オブジェクト種別]` | オブジェクト種別のテンプレートを表示 | | `-v [オブジェクト種別]` | オブジェクト種別の詳細テンプレートを表示 | | `--verbose` | 接続先など動作の詳細を表示 | | `--help` | ヘルプを表示 | | `--version` | バージョンを表示 |

### 仕様 - 1接続 1クエリ 1応答 (状態管理無し) - クエリ : 1行のクエリ + CRLF - 応答フォーマット : 規定なし - 文字コード : 規定なし (ASCII/UTF-8/Latin-1 などが混在) - 認証・暗号化 : 無し (平文) - アクセス制御 : 規定なし(レート制限等は各運営者が実施) - 主な情報源 : レジストリ(ドメイン)や RIR(IPアドレス・AS番号の地域レジストリ)が運用する WHOIS サーバ
