Docker (ドッカー)

  • コンテナ仮想化技術を利用したオープンプラットフォーム
  • アプリケーションの開発、デプロイ、実行を効率化出来る

メリット

  • 仮想環境であるため、隔離されており、実機に影響を及ぼしにくい
  • VMよりも軽量
  • 起動や動作が速い
  • 開発環境の共有や移行が容易

基本コマンド

コマンド 説明 
docker compose up --build -d
 Dockerfileに基づいてサービス用のDockerイメージを更新し、
バックグラウンドで全てのサービスを起動 
docker compose up -d
 バックグラウンドで全てのサービスを起動 
docker compose up
 全てのサービスをフォアグラウンドで起動し、ログを表示 
docker compose build
 Dockerfileに基づいてサービス用のDockerイメージをビルド 
docker compose exec [サービス名] [コマンド]
 既に実行中のコンテナ内で、新しいコマンドを実行する 
docker compose run [サービス名] [コマンド]
 一度限りのコマンドを実行 
docker compose down
 完全にコンテナを終了させる

-n [行数] : 最後の[行数]表示
-f : 追従

compose.yml 
# ==========================================
# compose.yml
#
# | 項目 | 内容 | 備考 |
# | --- | --- | --- |
# | name | プロジェクト名 | 全て小文字 |
# | services | 作成したいサービスを列挙 | YAMLはスペース2つ |
# | linux / db | 作成するサービスの名前 | docker composeで管理する際に必要 |
#
# ==========================================

# name: project1

services:
  linux:
    # ==========================================
    # linuxサービスの設定項目
    #
    # | 項目 | 内容 | 備考 |
    # | --- | --- | --- |
    # | container_name | コンテナ名 | 外部から見た時の名前 |
    # | hostname | ホスト名 | - |
    # | build | Dockerfileの設定 | - |
    # | image | 使用するコンテナイメージ | Dockerfileが無い場合指定 |
    # | volumes | マウントするフォルダ設定(データを永続化) | ホストディレクトリ: コンテナ内ディレクトリ |
    # | restart | コンテナの起動設定 | no unless-stopped always ... |
    # | environment | コンテナ内の環境変数を設定 | Python の `os.environ.get` 等でも取得可能 |
    # | ports | 外部からアクセス可能なポートを設定 | ホストマシンのポート番号:コンテナのポート番号 |
    # | depends_on | サービス間の依存関係を定義 | - |
    # | command | サービスが起動したときに実行するコマンド | - |
    # 
    # IPアドレスの確認 : docker inspect --format="{{range .NetworkSettings.Networks}}{{.IPAddress}}{{end}}" [${container_name}の値]
    #
    # ==========================================
    env_file: .env
    
    container_name: ${CONTAINER_NAME}

    hostname: ${CONTAINER_NAME}

    build: 
      # ==========================================
      # Dockerfileの設定項目
      #
      # | 項目 | 内容 |
      # | --- | --- |
      # | context/dockerfile | Dockerfileへの相対パス |
      # | args | Dockerfile内の環境変数を設定 |
      #
      # ==========================================

      context: ${Dockerfile_DIR}
      dockerfile: ${Dockerfile}
      args:
        WS:           ${WS}
        USER_NAME:    ${USER_NAME}
        PASSWORD:     ${PASSWORD}
        ENTRY_DIR:    ${ENTRY_DIR}
        ENTRY_POINT:  ${ENTRY_POINT}
        LANG:         ${LANG}
    
    # image: ubuntu:25.10

    volumes:
      - ${VOLUME}:/home/${USER_NAME}/${WS}

    restart: no

    environment:
      - TZ=${TIME_ZONE}
      - LANG=${LANG}
    
    ports:
      - "${PORT}:3000"
    
    # depends_on:
      # ==========================================
      # サービス間 依存関係の設定項目
      #
      # | 項目 | 内容 | 備考 |
      # | --- | --- | --- |
      # | db | 依存サービス名 | dbサービス参照 |
      #
      # ==========================================
      # - db                                # 先にdbサービスが動いていることを確認する

    command:
      - /home/${USER_NAME}/${ENTRY_DIR}/${ENTRY_POINT}
    
    networks:
      - docker_eth

networks:
  default:
    external: true
    name: bridge

  docker_eth:
    name: ${NETWORK_NAME}
    driver: bridge
    ipam:
      driver: default
      config:                               # 物理ネットワークに直接接続したい場合は実環境に合わせる
        - subnet: ${IP_SUBNET_ADDR}/${IP_PREFIX}
          gateway: ${IP_GATEWAY_ADDR}

DISP_STR(){
    for arg in "$@"; do
        printf "%s" "${arg}"
    done
}

trim() {
    local var="$1"
    var="${var#"${var%%[![:space:]]*}"}"   # 前
    var="${var%"${var##*[![:space:]]}"}"   # 後
    echo "$var"
}

# ===== ログ出力用 =====
LOG(){
    local name=$(trim "${1:-USERAPP}")
    local state=$(trim "${2:-LOG}")
    local action=$(trim "${3:-ACTION}")
    local nowtime="$(date '+%Y/%m/%d %H:%M:%S')"
    local detail=$(trim "${4:-log detail}")
    local header="[${name}][${state}][${action}]"
    local headerSize="${5:-50}"
    
    printf "%-${headerSize}s %s %s\n" "${header}" "${nowtime}" "${detail}"
}

LOG_INFO(){

    local state="INFO"
    local action="${2:-ACTION}"
    local detail="${3:-detail}"
    local headerSize="${4:-50}"

# ===== フールプルーフ =====
BLOCKED_COMMANDS=("chmod" "chown" "rm")
BLOCKED_DIRS=("/")
ROOT_CHILD_LIMIT=3

_guard_check() {
    local cmd_name
    cmd_name="$(basename -- "$1")"
    shift

    local blocked=1 c
    for c in "${BLOCKED_COMMANDS[@]}"; do
        [[ "$cmd_name" == "$c" ]] && blocked=0
    done
    (( blocked == 0 )) || return 0

    local arg resolved bd parent
    local end_opts=0 root_children=0

    for arg in "$@"; do
        if [[ "$arg" == "--no-preserve-root" ]]; then
            echo "[Reject] --no-preserve-root は使用禁止 ($cmd_name)" >&2
            return 1
        fi
        if (( end_opts == 0 )); then
            [[ "$arg" == "--" ]] && { end_opts=1; continue; }
            [[ "$arg" == -* ]] && continue
        fi

        # 存在しないパスも正規化できるよう -m を使う (GNU coreutils)
        resolved="$(realpath -m -- "$arg" 2>/dev/null)" || continue

        for bd in "${BLOCKED_DIRS[@]}"; do
            if [[ "$resolved" == "$bd" ]]; then
                echo "[Reject] '$arg' は '$bd' に解決されます (pwd=$PWD, cmd=$cmd_name)" >&2
                return 1
            fi
        done

        # `/*` はシェルが展開してから渡されるため、/ 直下が大量に並ぶ形で検出する
        parent="$(dirname -- "$resolved")"
        if [[ "$parent" == "/" && "$resolved" != "/" && -d "$resolved" ]]; then
            root_children=$((root_children + 1))
        fi
    done

    if (( root_children >= ROOT_CHILD_LIMIT )); then
        echo "[Reject] '/' 直下のディレクトリが ${root_children} 個指定されています (/* 展開の疑い, cmd=$cmd_name)" >&2
        return 1
    fi
    return 0
}

sudo() {
    local args=("$@") i=0
    while (( i < ${#args[@]} )); do
        case "${args[i]}" in
            -u|-g|-C|-D|-h|-p|-R|-r|-T|-t|-U) i=$((i + 2)) ;;
            -*) i=$((i + 1)) ;;
            *) break ;;
        esac
    done
    if (( i < ${#args[@]} )); then
        _guard_check "${args[@]:i}" || return 1
    fi
    command sudo "$@"
}

for _guard_cmd in "${BLOCKED_COMMANDS[@]}"; do
    eval "${_guard_cmd}() { _guard_check ${_guard_cmd} \"\$@\" || return 1; command ${_guard_cmd} \"\$@\"; }"
done
unset _guard_cmd

EOF
